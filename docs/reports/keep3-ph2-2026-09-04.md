# 取り込んだ発音が、単語シートの保存で推測に置き換わる ── 走らせた台本と出た答え

`claude/keep3`、2026-09-04。`docs/EXPIRY.md` 3-a-2。

**これは検査ではありません。**本物のアプリに一つだけ質問した台本です。
`www/wordsheet.js` は `claude/pop` のものなので直せず、直せないまま
`migrate-check` に足すと赤が一本ゲートに残るため、ここに置いてあります。
**直す人は、そのまま `tools/migrate-check.mjs` へ写せます。**

## 出た答え

```
取り込んだまま、起動しただけ        ph = "t sʰ ɑ ŋ"
単語シートで保存を押した（landed = true）
  そのすぐあと                     ph = ""
  次の起動のあと                   ph = "k a n o"

取り込んだ発音: "t sʰ ɑ ŋ"
→ 置き換わった。phGuess("kano") がアプリの推測。
```

`spv:1` を付けてあるので `migrateSp()` はこの単語に何もしません。
**3-a とは別の道であることの確かめ方です。**

**実機では押していません。**

## 台本

`tools/browser.mjs` を絶対パスで読んでいます。写すときは
`migrate-check.mjs` の書き方（`./browser.mjs`）に直してください。

```js
/* Not a check and not committed: one question asked of the real app.
   Does an imported pronunciation survive pressing Save on the word sheet? */
import http from 'http';
import fs from 'fs';
import path from 'path';
import { chromium, LAUNCH } from '/home/user/LLLLLLL/tools/browser.mjs';

const WWW = '/home/user/LLLLLLL/www';
const PORT = 8177;
const MIME = { '.html': 'text/html', '.js': 'text/javascript', '.css': 'text/css' };
const srv = http.createServer((q, r) => {
  const f = path.join(WWW, q.url === '/' ? 'index.html' : q.url.split('?')[0]);
  let body;
  try { body = fs.readFileSync(f); } catch (e) { r.writeHead(404); r.end(); return; }
  r.writeHead(200, { 'Content-Type': MIME[path.extname(f)] || 'text/plain', 'Cache-Control': 'no-store' });
  r.end(body);
}).listen(PORT);

const br = await chromium.launch(LAUNCH);
const pg = await br.newPage();
await pg.goto(`http://localhost:${PORT}/`);
const settle = () => pg.waitForTimeout(150);

const PH = ['t', 'sʰ', 'ɑ', 'ŋ'];
await pg.evaluate((ph) => {
  localStorage.clear();
  localStorage.setItem('lingua.langs', JSON.stringify({ LP: { name: 'Imported', mine: true } }));
  localStorage.setItem('lingua.cur', 'LP');
  localStorage.setItem('lingua.LP.lang', 'Imported');
  localStorage.setItem('lingua.LP.letters', JSON.stringify(
    [{ id: 'pk', nm: 'k', snd: ['k'] }, { id: 'pa', nm: 'a', snd: ['a'] },
     { id: 'pn', nm: 'n', snd: ['n'] }, { id: 'po', nm: 'o', snd: ['o'] }]));
  /* exactly what www/import.js:818 leaves behind: a headword, a spelling and
     the pronunciation column the person filled in. spv:1 so migrateSp has
     nothing to say about it -- this is the OTHER door. */
  localStorage.setItem('lingua.LP.words', JSON.stringify([
    { hw: 'kano', mn: 'mountain', mns: ['mountain'], pos: 'n', spv: 1,
      sp: [{ l: 'pk' }, { l: 'pa' }, { l: 'pn' }, { l: 'po' }],
      ph: ph }]));
}, PH);

const phOf = () => pg.evaluate(() => {
  const raw = JSON.parse(localStorage.getItem(langKey('words')) || '[]');
  const w = raw[0] || {};
  return { hw: String(w.hw || ''), ph: (w.ph || []).join(' ') };
});

await pg.reload(); await settle();
console.log('取り込んだまま、起動しただけ        ph =', JSON.stringify((await phOf()).ph));

/* Somebody opens the word and presses Save. Nothing about the pronunciation
   is on that screen; they changed a meaning, or they changed nothing. */
const landed = await pg.evaluate(() => {
  openHw = 'kano';
  const w = findWord('kano');
  wEdit = { seq: [], sp: JSON.parse(JSON.stringify(w.sp)), mns: (w.mns || []).slice(),
            pos: w.pos || '', ex: [] };
  return wdWrite();
});
console.log('単語シートで保存を押した（landed =', landed + '）');
console.log('  そのすぐあと                     ph =', JSON.stringify((await phOf()).ph));

await pg.reload(); await settle();
const after = await phOf();
console.log('  次の起動のあと                   ph =', JSON.stringify(after.ph));
console.log();
console.log('取り込んだ発音:', JSON.stringify(PH.join(' ')));
console.log(after.ph === PH.join(' ')
  ? '→ 残っている'
  : '→ 置き換わった。phGuess("' + after.hw + '") がアプリの推測。');
await br.close();
srv.close();
```
